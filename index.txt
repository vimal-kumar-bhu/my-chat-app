    <script type="module">
        import { initializeApp } from "https://www.gstatic.com/firebasejs/10.8.0/firebase-app.js";
        import { getDatabase, ref, push, onChildAdded } from "https://www.gstatic.com/firebasejs/10.8.0/firebase-database.js";

        // Aapki Firebase Configuration Keys
        const firebaseConfig = {
            apiKey: "AIzaSyD9Z8wZE50quV3YER_fYRWyMGOx4nb16J8",
            authDomain: "my-realtime-chat-1070f.firebaseapp.com",
            databaseURL: "https://my-realtime-chat-1070f-default-rtdb.firebaseio.com",
            projectId: "my-realtime-chat-1070f",
            storageBucket: "my-realtime-chat-1070f.firebasestorage.app",
            messagingSenderId: "164361933197",
            appId: "1:164361933197:web:2c1d02d7c9ceb9fc182dc4"
        };

        const app = initializeApp(firebaseConfig);
        const db = getDatabase(app);

        // User ka naam pooch lo
        let myName = prompt("Apna naam enter karein:") || "User";
        document.getElementById('usernameDisplay').innerText = myName;
        document.getElementById('userAvatar').innerText = myName.charAt(0).toUpperCase();

        const form = document.getElementById('chatForm');
        const input = document.getElementById('messageInput');
        const container = document.getElementById('chatContainer');

        // Message Send Karna
        form.addEventListener('submit', (e) => {
            e.preventDefault();
            const text = input.value.trim();
            if(!text) return;

            const messagesRef = ref(db, 'messages');
            push(messagesRef, {
                name: myName,
                text: text,
                time: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })
            });
            input.value = '';
        });

        // Real-time Messages Sunna
        const messagesRef = ref(db, 'messages');
        onChildAdded(messagesRef, (snapshot) => {
            const data = snapshot.val();
            const isMe = data.name === myName;
            
            if(isMe) {
                container.innerHTML += `
                    <div class="flex justify-end">
                        <div class="bg-indigo-600 text-white rounded-xl rounded-tr-none p-2.5 max-w-[80%]">
                            <p class="text-[10px] font-bold text-indigo-200">Aap</p>
                            <p class="text-xs">${data.text}</p>
                            <span class="text-[9px] text-indigo-200 mt-0.5 block text-right">${data.time}</span>
                        </div>
                    </div>
                `;
            } else {
                container.innerHTML += `
                    <div class="flex justify-start">
                        <div class="bg-slate-800 border border-slate-700/50 rounded-xl rounded-tl-none p-2.5 max-w-[80%]">
                            <p class="text-[10px] font-bold text-emerald-400">${data.name}</p>
                            <p class="text-xs text-slate-200">${data.text}</p>
                            <span class="text-[9px] text-slate-400 mt-0.5 block text-right">${data.time}</span>
                        </div>
                    </div>
                `;
            }
            container.scrollTop = container.scrollHeight;
        });
    </script>
